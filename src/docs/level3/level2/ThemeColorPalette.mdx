import { Meta, ColorPalette, ColorItem } from '@storybook/addon-docs/blocks';
import { THEMES } from '../../../stories/Theme';

export const capitalize = (s) => s.charAt(0).toUpperCase() + s.slice(1);

export const isObject = (v) => typeof v === 'object' && v !== null;

export const isColors = (node) => Object.values(node).every((v) => typeof v === 'string');

export const Tree = ({ node, title, depth = 2 }) => {
  const Heading = `h${Math.min(depth, 6)}`;
  const children = Object.entries(node).filter(([, v]) => isObject(v));
  const tokens = children.filter(([, v]) => isColors(v));
  const branches = children.filter(([, v]) => !isColors(v));

  return (
    <section>
      <Heading>{title}</Heading>

      {tokens.length > 0 && (
        <ColorPalette>
          {tokens.map(([key, colors]) => (
            <ColorItem
              key={key}
              title={key}
              subtitle={Object.keys(colors).join(' · ')}
              colors={colors}
            />
          ))}
        </ColorPalette>
      )}

      {branches.map(([key, child]) => (
        <Tree key={key} node={child} title={child.name ?? capitalize(key)} depth={depth + 1} />
      ))}
    </section>
  );
};

<Meta title="Documentation/Theme/ColorPalette" />

# Color Palette

{Object.entries(THEMES).map(([key, theme]) => (
  <Tree key={key} node={theme} title={theme.name} />
))}
